import concurrent.futures, json, pathlib, re, subprocess
adb = str(pathlib.Path.home() / 'Library/Android/sdk/platform-tools/adb')
out = pathlib.Path('build/main-verification')
def run(api, serial, variant):
    suffix = ".developer" if variant=="developer" else ""
    expected = 204 if variant=="developer" else 145
    cmds = [
        ['install','-r','-t',f'app/build/outputs/apk/{variant}/debug/app-{variant}-debug.apk'],
        ['install','-r','-t',f'app/build/outputs/apk/androidTest/{variant}/debug/app-{variant}-debug-androidTest.apk'],
        ['shell','input','keyevent','KEYCODE_WAKEUP'],
        ['shell','wm','dismiss-keyguard'],
        ['shell','am','instrument','-w','-r','-e','helix.smoke.model','qwen2.5:3b-instruct','-e','helix.smoke.ollamaUiModel','qwen3-vl:2b-instruct',f'com.helix.agent{suffix}.test/com.helix.app.HelixAndroidJUnitRunner'],
    ]
    log = out / f'app-{variant}-api{api}-settled.log'
    print('START', api, variant, flush=True)
    with log.open('w') as f:
        for cmd in cmds:
            p = subprocess.run([adb,'-s',serial]+cmd,stdout=f,stderr=subprocess.STDOUT)
            f.flush()
            if p.returncode: raise RuntimeError(f'{api} command failed: {cmd}')
    body = log.read_text()
    record = {}; results = []
    for line in body.splitlines():
        m = re.match(r'INSTRUMENTATION_STATUS: (class|test)=(.*)', line)
        if m: record[m[1]] = m[2]
        m = re.match(r'INSTRUMENTATION_STATUS_CODE: (-?\d+)$', line)
        if m and int(m[1]) in (0,-1,-2,-3,-4):
            results.append(dict(record, code=int(m[1]))); record={}
    counts = {str(code):sum(r['code']==code for r in results) for code in (0,-1,-2,-3,-4)}
    passed = bool(re.search(r'^OK \(\d+ tests?\)', body, re.M)) and not any(x in body for x in ['FAILURES!!!','INSTRUMENTATION_FAILED','Process crashed'])
    passed = passed and len(results)==expected and len({(r.get("class"),r.get("test")) for r in results})==expected and counts['-1']==0 and counts['-2']==0
    data = dict(api=api,expected=expected,observed=len(results),counts=counts,passed=passed,results=results)
    (out/f'app-{variant}-api{api}-settled.json').write_text(json.dumps(data, indent=2))
    print('END',api,variant,counts,passed,flush=True)
    return passed
def matrix(api, serial):
    return all([run(api,serial,'consumer'),run(api,serial,'developer')])
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    results = [pool.submit(matrix,29,'<OWNED_EMULATOR_SERIAL>'),pool.submit(matrix,36,'<OWNED_EMULATOR_SERIAL>')]
    assert all(f.result() for f in results), 'full application instrumentation failed'
