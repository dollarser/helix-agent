@RunWith(AndroidJUnit4::class)
class GoalCompletionDeviceTest {
    private val clock =
        object : Clock {
            override fun now(): Instant = Instant.ofEpochMilli(3000)
        }

    @Test
    fun localToolCompletionIsAtomicAndRepeatedSettlementIsIdempotent() =
        fixture { storage, verifier, _ ->
            bind(storage, CriterionVerificationMethod.LOCAL_TOOL_SUCCESS, "time.now")
            val settlement = GoalRunSettlement(storage, clock) { UUID.randomUUID().toString() }
            settlement.settle("turn", verifier::refresh)
            val goal = storage.goals.resolve("goal")
            val run = storage.goalRuns.resolve("run")
            val audit = storage.auditEvents.listByCorrelation("corr")
            assertEquals("COMPLETED", goal.state)
            assertEquals("COMPLETED", run.outcome)
            assertNotNull(
                goal.criteria
                    .single()
                    .evidence
                    ?.verification
                    ?.sourceHash,
            )
            settlement.settle("turn", verifier::refresh)
            assertEquals(goal, storage.goals.resolve("goal"))
            assertEquals(run, storage.goalRuns.resolve("run"))
            assertEquals(audit, storage.auditEvents.listByCorrelation("corr"))
        }

    @Test
    fun literalCompletionCapturesAnActualToolResultArtifact() =
        fixture { storage, verifier, _ ->
            bind(storage, CriterionVerificationMethod.ARTIFACT_UTF8_CONTAINS, "真实输出")
            GoalRunSettlement(storage, clock) { UUID.randomUUID().toString() }.settle("turn", verifier::refresh)
            val goal = storage.goals.resolve("goal")
            assertEquals("COMPLETED", goal.state)
            assertNotNull(
                goal.criteria
                    .single()
                    .evidence
                    ?.artifactRef,
            )
            assertEquals(1, storage.artifacts.listBySession("session").size)
        }

    @Test
    fun missingContentAndUnsettledCallsCannotComplete() =
        fixture { storage, verifier, _ ->
            bind(storage, CriterionVerificationMethod.LOCAL_TOOL_SUCCESS, "time.now")
            val result = requireNotNull(storage.toolResults.byToolCall("call"))
            storage.contentStore.delete(ContentRef.parse(requireNotNull(result.contentRef)))
            GoalRunSettlement(storage, clock) { UUID.randomUUID().toString() }.settle("turn", verifier::refresh)
            assertEquals("PAUSED", storage.goals.resolve("goal").state)
            assertEquals("RUN_FINISHED", storage.goalRuns.resolve("run").outcome)
        }

    @Test
    fun pendingCallBlocksCompletionDespiteSuccessfulSelectedCall() =
        fixture { storage, verifier, _ ->
            bind(storage, CriterionVerificationMethod.LOCAL_TOOL_SUCCESS, "time.now")
            storage.toolCalls.append("pending", "turn", "pending", "time.now", "1", "{}", "PENDING")
            GoalRunSettlement(storage, clock) { UUID.randomUUID().toString() }.settle("turn", verifier::refresh)
            assertEquals("BLOCKED", storage.goals.resolve("goal").state)
            assertEquals("BLOCKED(NEEDS_REVIEW)", storage.goalRuns.resolve("run").outcome)
        }

    @Test
    fun rollbackLeavesGoalRunAndAuditUncompleted() =
        fixture { storage, verifier, _ ->
            bind(storage, CriterionVerificationMethod.LOCAL_TOOL_SUCCESS, "time.now")
            val before = storage.goals.resolve("goal")
            assertThrows(IllegalStateException::class.java) {
                storage.withTransaction {
                    GoalRunSettlement(storage, clock) { UUID.randomUUID().toString() }.settle("turn", verifier::refresh)
                    error("simulated terminal transaction failure")
                }
            }
            assertEquals(before, storage.goals.resolve("goal"))
            assertNull(storage.goalRuns.resolve("run").endedAt)
            assertEquals(emptyList<Any>(), storage.auditEvents.listByCorrelation("corr"))
        }

    @Test
    fun explicitContinueConsumesManualReviewInsideTurnTerminalTransaction() =
        fixture { storage, completion, workspace ->
            bind(storage, CriterionVerificationMethod.MANUAL_REVIEW, "")
            val goal = storage.goals.resolve("goal").toRuntimeGoal()
            storage.goalRuns.finish(storage.goalRuns.resolve("run"), "RUN_FINISHED", 2000, 0, 0, 0, 0)
            storage.goals.updateGoal(goal.copy(state = GoalState.PAUSED).toStoredGoal())
            val reader = goalEvidenceReader(storage)
            val criterion = goal.criteria.single()
            val selection =
                GoalEvidenceReview(
                    "call",
                    requireNotNull(criterion.binding).hash(criterion.id, criterion.description),
                    reader.read("goal", "call").hash,
                )
            val verifier = GoalCriterionVerifier(reader, GoalToolArtifactStore(storage, workspace, reader), clock)
            GoalCriterionEditor(
                storage,
                verifier,
                clock,
            ) { UUID.randomUUID().toString() }.stageReview("goal", "c1", selection)
            val started =
                requireNotNull(
                    GoalRunCoordinator(storage, clock) { UUID.randomUUID().toString() }.start(
                        GoalTurnStart(
                            "goal",
                            GoalWakeReason.USER_OPEN,
                            TurnStartSpec("session", "next", "next-model", "snapshot", "Continue"),
                            TurnBudgets(5, 10, 800, 800, 5000),
                        ),
                    ),
                )
            started.coordinator.beginModelStream()
            started.coordinator.terminalize(ModelStreamTerminal(TurnState.COMPLETED, null), completion::refresh)
            val completed = storage.goals.resolve("goal")
            assertEquals("COMPLETED", completed.state)
            assertEquals("COMPLETED", storage.turns.resolve("next").state)
            assertNull(completed.criteria.single().pendingReview)
            assertNotNull(completed.criteria.single().evidence)
            assertEquals("RUN_FINISHED", storage.goalRuns.resolve("run").outcome)
            assertEquals("COMPLETED", storage.goalRuns.resolve(started.runId).outcome)
        }

    private fun bind(
        storage: HelixStorage,
        method: CriterionVerificationMethod,
        argument: String,
    ) {
        val goal = storage.goals.resolve("goal").toRuntimeGoal()
        val criterion =
            goal.criteria.single().withBinding(
                "Verify actual result",
                CriterionVerificationBinding(method, argument),
            )
        storage.goals.updateGoal(goal.copy(criteria = listOf(criterion)).toStoredGoal())
    }

    private fun fixture(block: (HelixStorage, GoalCompletionVerifier, File) -> Unit) =
        withGoalEvidenceStorage { storage ->
            seedGoalToolEvidence(storage)
            val context = ApplicationProvider.getApplicationContext<Context>()
            val workspace = File(context.cacheDir, "goal-completion-${UUID.randomUUID()}")
            val registry = ToolRegistry(listOf(BuiltInToolSource(listOf(TimeNowTool.descriptor()))))
            val verifier = GoalCompletionVerifier(storage, registry, workspace, clock) { UUID.randomUUID().toString() }
            try {
                block(storage, verifier, workspace)
            } finally {
                workspace.deleteRecursively()
            }
        }
}
