/** Opt-in real model, real write Dispatcher/approval and production Goal completion through UI. */
@RunWith(AndroidJUnit4::class)
class GoalRealCompletionUiTest {
    @get:Rule val compose = createAndroidComposeRule<MainActivity>()
    private val container get() = compose.container()
    private val storage get() = container.storage
    private val service get() = container.chatService
    private val marker = "GOAL_ARTIFACT_${UUID.randomUUID()}"
    private val relative = "output/$marker.txt"
    private val toolName get() =
        InstrumentationRegistry.getArguments().getString("helix.goalCompletion.tool")
            ?: "write"
    private val resultSuffix get() =
        if (toolName == "write") {
            ""
        } else if (toolName == "edit") {
            "-edit"
        } else {
            "-proot"
        }
    private var providerId: String? = null
    private var sessionId: String? = null
    private var goalId: String? = null

    @Test fun realModelWritesApprovedArtifactAndCompletesGoal() =
        runBlocking {
            val args = InstrumentationRegistry.getArguments()
            assumeTrue(args.getString("helix.goalCompletion") == "true")
            require(toolName in setOf("write", "edit", "code.linux.run"))
            val model = requireNotNull(args.getString("helix.goalCompletion.model"))
            val port = requireNotNull(args.getString("helix.goalCompletion.port")).toInt()
            val protocol =
                ProviderProtocol.parse(args.getString("helix.goalCompletion.protocol") ?: "OPENAI_CHAT_COMPLETIONS")
            val previous = service.runControl.value
            val previousProfile = container.profileStore.profile
            try {
                compose.resetDeterministicUiState()
                if (toolName == "code.linux.run") {
                    require(com.helix.app.proot.ProotToolModule.AVAILABLE)
                    container.profileStore.switchTo(com.helix.core.model.SafetyProfile.ADVANCED)
                    com.helix.app.proot.ProotToolModule
                        .verifyNowNote()
                }
                createProvider(model, port, protocol)
                sessionId = service.createSession("Real Goal completion", requireNotNull(providerId), model)
                service.openSession(requireNotNull(sessionId))
                service.setTurnBudgets(TurnBudgets(8, 4, 65536, 4096, 69632))
                compose.navigateTo("sessions")
                compose.waitUntil(10000) { service.screen.value.openSessionId == sessionId }
                compose.onNodeWithTag("chat-mode-menu").performClick()
                compose.onNodeWithTag("chat-mode-goal").performClick()
                createBoundGoal()
                compose.onNodeWithTag("goal-continue-$goalId").performScrollTo().performClick()
                approveWrite()
                compose.waitUntil(180000) { storage.goals.resolve(requireNotNull(goalId)).state == "COMPLETED" }
                verifyAndRecord(model, protocol)
            } finally {
                service.stop()
                compose.waitUntil(10000) { !service.screen.value.isSending }
                service.closeSession()
                container.profileStore.switchTo(previousProfile)
                service.setMode(previous.mode)
                service.setTurnBudgets(previous.budgets)
                goalId?.let(storage.goals::delete)
                sessionId?.let { storage.sessions.archive(it, System.currentTimeMillis()) }
                providerId?.let { container.providerService.delete(it) }
                File(compose.activity.filesDir, "workspaces/app/$relative").delete()
            }
        }

    private suspend fun createProvider(
        model: String,
        port: Int,
        protocol: ProviderProtocol,
    ) {
        require(port in 1..65535)
        val draft =
            ProviderDraft(
                null,
                "Goal completion ${UUID.randomUUID()}",
                protocol,
                NormalizedEndpoint.parse("http://10.0.2.2:$port/v1"),
                model,
                "{}",
                false,
                CleartextAuthorization("10.0.2.2", port),
                emptyList(),
            )
        providerId = container.providerService.create(draft, null, cleartextConfirmed = true)
        check(container.providerService.runConnectionTest(requireNotNull(providerId)) is ProbeOutcome.Ok)
    }

    private fun createBoundGoal() {
        val objective =
            if (toolName == "code.linux.run") {
                "Use code.linux.run exactly once with script: printf %s $marker > result.txt . " +
                    "The script must end at result.txt, without that final punctuation. " +
                    "Do not set output or use other tools. After the approved command succeeds, reply DONE."
            } else if (toolName == "edit") {
                val before = "BEFORE_EDIT_$marker"
                val file = File(compose.activity.filesDir, "workspaces/app/$relative")
                requireNotNull(file.parentFile).mkdirs()
                file.writeText(before)
                val hash = FileContentStore.sha256Hex(before.toByteArray())
                "Use edit exactly once on scope:app:$relative, oldText $before, newText $marker, " +
                    "expectedSha256 $hash. Do not use other tools. After the approved edit succeeds, reply DONE."
            } else {
                "Use write exactly once to create scope:app:$relative with exact content $marker " +
                    "and no trailing newline. Do not use other tools. After the approved write succeeds, reply DONE."
            }
        compose.onNodeWithTag("chat-input").performTextInput(objective)
        compose.onNodeWithTag("goal-manage").performClick()
        compose.onNodeWithTag("goal-create").performClick()
        compose.onNodeWithTag("goal-criteria").performTextInput("The written snapshot has the expected content hash.")
        compose.onNodeWithTag("goal-save").performClick()
        compose.waitUntil(10000) {
            goalId =
                storage.goals
                    .list()
                    .singleOrNull { it.objective == objective }
                    ?.id
            goalId != null
        }
        bindHash()
        assertEquals("READY", storage.goals.resolve(requireNotNull(goalId)).state)
    }

    private fun bindHash() {
        val criterionId =
            storage.goals
                .resolve(requireNotNull(goalId))
                .criteria
                .single()
                .id
        compose.onNodeWithTag("goal-criteria-$goalId").performScrollTo().performClick()
        compose.onNodeWithTag("goal-bind-$criterionId").performScrollTo().performClick()
        compose.onNodeWithTag("criterion-method-ARTIFACT_SHA256").performScrollTo().performClick()
        compose
            .onNodeWithTag("criterion-argument")
            .performScrollTo()
            .performTextInput(FileContentStore.sha256Hex(marker.toByteArray()))
        compose.onNodeWithTag("criterion-save").performClick()
        compose.waitUntil(10000) {
            storage.goals
                .resolve(requireNotNull(goalId))
                .criteria
                .single()
                .binding != null
        }
        // Close the criteria surface; the Goal manager stays open for explicit Continue.
        compose.onNodeWithTag("criteria-close").performClick()
    }

    private fun approveWrite() {
        compose.waitUntil(180000) {
            service.screen.value.toolTimeline
                .any { it.card?.state == ApprovalCardState.PENDING }
        }
        val row =
            service.screen.value.toolTimeline
                .single { it.card?.state == ApprovalCardState.PENDING }
        assertEquals(toolName, row.toolName)
        assertTrue(row.requestSummary.contains(marker))
        if (toolName != "code.linux.run") assertTrue(row.requestSummary.contains(relative))
        val tag = "approval-approve-${requireNotNull(row.card).approvalId}"
        compose.onNodeWithTag("chat-timeline").performScrollToNode(hasTestTag(tag))
        compose.onNodeWithTag(tag).assertIsDisplayed().performClick()
    }

    private fun verifyAndRecord(
        model: String,
        protocol: ProviderProtocol,
    ) {
        val goal = storage.goals.resolve(requireNotNull(goalId))
        val evidence = requireNotNull(goal.criteria.single().evidence)
        val artifact = storage.artifacts.resolve(requireNotNull(evidence.artifactRef).value)
        val call = storage.toolCalls.resolve(requireNotNull(evidence.toolCallId).value)
        assertEquals(toolName, call.name)
        assertEquals("COMPLETED", call.state)
        if (toolName == "code.linux.run") {
            assertTrue(artifact.id.startsWith("goal-archive-"))
        } else {
            assertEquals(marker, File(compose.activity.filesDir, "workspaces/app/$relative").readText())
        }
        assertEquals(marker, File(compose.activity.filesDir, "workspaces/app/${artifact.relativePath}").readText())
        assertTrue(goal.modelCalls > 0 && goal.totalTokens > 0)
        assertEquals(
            "COMPLETED",
            storage.goalRuns
                .listByGoal(goal.id)
                .single()
                .outcome,
        )
        compose.onNodeWithTag("goal-manage").performClick()
        compose
            .onNodeWithTag("goal-state-$goalId")
            .performScrollTo()
            .assertIsDisplayed()
            .assertTextEquals(compose.activity.getString(R.string.goal_state_completed))
        captureUi("completed-${protocol.name}$resultSuffix")
        compose.onNodeWithTag("goal-close").performClick()
        val result =
            buildJsonObject {
                put("model", model)
                put("protocol", protocol.name)
                put("tool", toolName)
                put("goalState", goal.state)
                put("modelCalls", goal.modelCalls)
                put("toolCalls", goal.toolCalls)
                put("tokens", goal.totalTokens)
                put("artifactSha256", artifact.sha256)
                put(
                    "scope",
                    "UI create and hash binding, real model, explicit $toolName approval, " +
                        "real file and Goal completion",
                )
            }
        val directory = File(compose.activity.filesDir, "hxa102-goal-completion").apply { mkdirs() }
        directory.resolve("result-${protocol.name}$resultSuffix.json").writeText(result.toString())
    }

    private fun captureUi(name: String) {
        val directory = File(compose.activity.filesDir, "hxa102-goal-completion").apply { mkdirs() }
        val bitmap = requireNotNull(InstrumentationRegistry.getInstrumentation().uiAutomation.takeScreenshot())
        try {
            directory.resolve("$name.png").outputStream().use {
                check(bitmap.compress(android.graphics.Bitmap.CompressFormat.PNG, 100, it))
            }
        } finally {
            bitmap.recycle()
        }
        val goal = goalId?.let(storage.goals::resolve)
        val state =
            buildJsonObject {
                put("goalState", goal?.state)
                put("modelCalls", goal?.modelCalls)
                put("blocked", service.screen.value.blockedReason)
                put(
                    "tools",
                    service.screen.value.toolTimeline
                        .joinToString { "${it.toolName}:${it.stateLabel}" },
                )
            }
        directory.resolve("$name-state.json").writeText(state.toString())
    }
}
