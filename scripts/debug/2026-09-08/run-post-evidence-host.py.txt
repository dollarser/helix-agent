import json, subprocess, os, pathlib, hashlib, xml.etree.ElementTree as ET
root=pathlib.Path('.')
out=root/'build/main-verification'
cmd=json.loads((out/'post-proot-host-command.json').read_text())
files=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','-z']).decode().split('\0')
source=[p for p in sorted(set(files)) if p and (p.endswith(('.kt','.kts','.xml','.toml','.properties','.gradle')) or p.endswith('gradle.lockfile')) and pathlib.Path(p).is_file()]
def fingerprints():
 return {p:hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest() for p in source}
before=fingerprints()
(out/'post-evidence-host-source.json').write_text(json.dumps(before,indent=2))
env=os.environ.copy()
env['JAVA_HOME']='/opt/homebrew/opt/openjdk@17/libexec/openjdk.jdk/Contents/Home'
env['HELIX_CONNECTOR_ACCEPTANCE_DIR']=str((root/'app/build/outputs/hxa-125-samples').resolve())
env['HELIX_CONNECTOR_SAMPLE_ZIP']=str((root/'app/build/outputs/connector-handoff-672dc5a/inputs/connector-参考包.zip').resolve())
(out/'post-evidence-host-command.json').write_text(json.dumps(cmd,indent=2))
with (out/'post-evidence-host.log').open('w') as log:
 result=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT)
counts={'tests':0,'failures':0,'errors':0,'skipped':0}
tasks=[]
for task in cmd:
 if not task.startswith(':') or ':test' not in task: continue
 parts=task.split(':')[1:]
 directory=root.joinpath(*parts[:-1])/'build/test-results'/parts[-1]
 suites=list(directory.glob('TEST-*.xml'))
 assert suites,task
 item={'task':task,'suites':len(suites)}
 for key in counts:
  item[key]=sum(int(ET.parse(p).getroot().get(key,0)) for p in suites)
  counts[key]+=item[key]
 tasks.append(item)
summary={'exit':result.returncode,'unchangedSource':before==fingerprints(),'sourceFiles':len(source),'counts':counts,'tasks':tasks}
(out/'post-evidence-host-result.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps({k:v for k,v in summary.items() if k!='tasks'}),flush=True)
raise SystemExit(result.returncode)
