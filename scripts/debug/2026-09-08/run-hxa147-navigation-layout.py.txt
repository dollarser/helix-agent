import argparse,subprocess,pathlib,hashlib,json,re
p=argparse.ArgumentParser();p.add_argument('--serial',required=True);p.add_argument('--flavor',choices=['consumer','developer'],required=True);p.add_argument('--language',choices=['zh-CN','en'],required=True);p.add_argument('--output',type=pathlib.Path,required=True);a=p.parse_args();a.output.mkdir(exist_ok=False,parents=True)
adb='<ANDROID_SDK>/platform-tools/adb';pkg='com.helix.agent'+('.developer' if a.flavor=='developer' else '')
def run(*args,check=True):return subprocess.run([adb,'-s',a.serial,*args],capture_output=True,check=check,timeout=120)
r=dict(serial=a.serial,flavor=a.flavor,language=a.language,apks=[],passed=False)
try:
 for package,apk in [(pkg,f'app/build/outputs/apk/{a.flavor}/debug/app-{a.flavor}-debug.apk'),(pkg+'.test',f'app/build/outputs/apk/androidTest/{a.flavor}/debug/app-{a.flavor}-debug-androidTest.apk')]:
  run('install','-r','-t',apk);remote=run('shell','pm','path',package).stdout.decode().strip().removeprefix('package:');got=run('shell','sha256sum',remote).stdout.decode().split()[0];sha=hashlib.sha256(pathlib.Path(apk).read_bytes()).hexdigest();assert sha==got;r['apks'].append(dict(package=package,sha256=sha,installedSha256=got))
 r['display']={name:run('shell','wm',name).stdout.decode() for name in ['size','density']};r['fontScale']=run('shell','settings','get','system','font_scale').stdout.decode().strip()
 run('shell','run-as',pkg,'rm','-rf','cache/hxa147-layout')
 cmd=[adb,'-s',a.serial,'shell','am','instrument','-w','-r','-e','class','com.helix.app.ui.NavigationLayoutDeviceTest','-e','helix.test.language',a.language,'-e','helix.layout.capture','true',pkg+'.test/com.helix.app.HelixAndroidJUnitRunner'];r['command']=cmd
 with (a.output/'instrumentation.log').open('w') as out:r['exitCode']=subprocess.run(cmd,stdout=out,stderr=subprocess.STDOUT,timeout=300).returncode
 log=(a.output/'instrumentation.log').read_text();codes=re.findall(r'^INSTRUMENTATION_STATUS_CODE: (-?\d+)$',log,re.M);r['codes']=codes;r['screenshots']={}
 for name in ['provider-picker','provider-form','files','browser','sessions']:
  data=run('exec-out','run-as',pkg,'cat',f'cache/hxa147-layout/{name}.png',check=False).stdout
  if data.startswith(b'\x89PNG'):(a.output/f'{name}.png').write_bytes(data);r['screenshots'][name]=hashlib.sha256(data).hexdigest()
 r['passed']=r['exitCode']==0 and codes.count('0')==1 and all(c in ['0','1'] for c in codes) and 'OK (1 test)' in log and len(r['screenshots'])==5
except Exception as e:r['error']=str(e)
finally:(a.output/'result.json').write_text(json.dumps(r,indent=2))
print(json.dumps({'passed':r['passed'],'output':str(a.output),'error':r.get('error')}),flush=True)
