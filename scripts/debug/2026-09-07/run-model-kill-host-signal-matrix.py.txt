import pathlib, subprocess, hashlib, json
root=pathlib.Path('build/main-verification/model-kill-api29-36-host-signal-fixed')
root.mkdir(exist_ok=False)
adb='<ANDROID_SDK>/platform-tools/adb'
paths={'com.helix.agent':pathlib.Path('app/build/outputs/apk/consumer/debug/app-consumer-debug.apk'),'com.helix.agent.test':pathlib.Path('app/build/outputs/apk/androidTest/consumer/debug/app-consumer-debug-androidTest.apk')}
records=[]
for api,serial in [(29,'<OWNED_EMULATOR_SERIAL>'),(36,'<OWNED_EMULATOR_SERIAL>')]:
 base=[adb,'-s',serial]
 subprocess.run(base+['install','-r',str(paths['com.helix.agent.test'])],check=True)
 hashes={}
 for pkg,p in paths.items():
  remote=subprocess.check_output(base+['shell','pm','path',pkg],text=True).strip().removeprefix('package:')
  actual=subprocess.check_output(base+['shell','sha256sum',remote],text=True).split()[0]
  expected=hashlib.sha256(p.read_bytes()).hexdigest()
  assert actual==expected,(pkg,actual,expected)
  hashes[pkg]=actual
 (root/f'api{api}-installed-apks.json').write_text(json.dumps(hashes,indent=2)+'\n')
 for protocol in ['OPENAI_CHAT_COMPLETIONS','OPENAI_RESPONSES','ANTHROPIC_MESSAGES']:
  for boundary in ['headers','body']:
   name=f'api{api}-{protocol.lower()}-{boundary}'
   with (root/f'{name}-host.log').open('w') as log:
    r=subprocess.run(['python3','scripts/run-model-stream-process-kill.py','--adb',adb,'--serial',serial,'--protocol',protocol,'--boundary',boundary,'--output',str(root/name)],stdout=log,stderr=subprocess.STDOUT,timeout=180)
   records.append(dict(api=api,protocol=protocol,boundary=boundary,exit=r.returncode))
   (root/'result.json').write_text(json.dumps(records,indent=2)+'\n')
   print(records[-1],flush=True)
   if r.returncode: raise SystemExit(r.returncode)
