  with (root/f'{name}-host.log').open('w') as log:
   r=subprocess.run(['python3',script,'--adb','<ANDROID_SDK>/platform-tools/adb','--serial',serial,'--output',str(root/name)],stdout=log,stderr=subprocess.STDOUT,timeout=180)
